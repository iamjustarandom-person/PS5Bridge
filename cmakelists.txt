cmake_minimum_required(VERSION 3.20)

project(PS5Bridge VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Vulkan REQUIRED)

add_executable(PS5Bridge src/main.cpp)

target_link_libraries(PS5Bridge PRIVATE Vulkan::Vulkan)

if(WIN32)
    target_link_libraries(PS5Bridge PRIVATE
        d3d11
        user32
        gdi32
    )
endif()
